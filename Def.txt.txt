// Настройка PDF.js worker
pdfjsLib.GlobalWorkerOptions.workerSrc = 'https://cdnjs.cloudflare.com/ajax/libs/pdf.js/3.11.174/pdf.worker.min.js';

class SmartGrader {
    constructor() {
        this.config = {
            lmStudioUrl: localStorage.getItem('lmStudioUrl') || 'http://localhost:1234',
            temperature: parseFloat(localStorage.getItem('temperature')) || 0.3,
            maxTokens: parseInt(localStorage.getItem('maxTokens')) || 4096
        };

        this.state = {
            currentFile: null,
            fileContent: null,
            selectedModel: null,
            selectedRubric: null,
            isAnalyzing: false,
            models: [],
            rubrics: JSON.parse(localStorage.getItem('rubrics') || '[]'),
            history: JSON.parse(localStorage.getItem('history') || '[]'),
            currentResult: null,
            expertResults: {},
            evaluationPlan: null
        };

        this.init();
    }

    init() {
        this.migrateRubrics();
        this.loadRubrics();
        this.loadHistory();
        this.checkConnection();

        if (this.state.rubrics.length === 0) {
            this.createDefaultRubric();
        }

        this.updateRubricStatus();
    }

    migrateRubrics() {
        let modified = false;
        this.state.rubrics = this.state.rubrics.map(r => {
            if (typeof r.id === 'number') {
                r.id = String(r.id);
                modified = true;
            }
            if (!r.criteria || !Array.isArray(r.criteria)) {
                r.criteria = [];
                modified = true;
            }
            return r;
        });

        if (modified) {
            this.saveRubrics();
        }
    }

    resetRubrics() {
        if (confirm('Удалить все рубрики и создать стандартную заново?')) {
            this.state.rubrics = [];
            this.state.selectedRubric = null;

    async processFile(file) {
        const extension = '.' + file.name.split('.').pop().toLowerCase();
        
        this.log(`📄 Загрузка файла: ${file.name} (${this.formatFileSize(file.size)})`, 'info');

        try {
            let content = null;
            let iconClass = 'fa-file-alt';
            let fileType = 'text';

            if (extension === '.pdf') {
                this.log('🔍 Извлечение текста из PDF...', 'info');
                content = await this.extractTextFromPDF(file);
                iconClass = 'fa-file-pdf text-red-400';
                fileType = 'pdf';
            } else if (extension === '.docx') {
                this.log('🔍 Извлечение текста из DOCX...', 'info');
                content = await this.extractTextFromDOCX(file);
                iconClass = 'fa-file-word text-blue-400';
                fileType = 'docx';
            } else {
                // Текстовые файлы
                const textExtensions = ['.txt', '.md', '.py', '.js', '.html', '.css', '.json', '.java', '.cpp', '.c', '.h', '.cs', '.php', '.rb', '.go', '.rs', '.swift', '.kt', '.ts', '.jsx', '.tsx', '.sql', '.xml', '.yaml', '.yml'];
                if (!textExtensions.includes(extension)) {
                    this.showToast(`Формат ${extension} не поддерживается. Используйте PDF, DOCX или текстовые файлы.`, 'error');
                    return;
                }
                content = await this.readFileContent(file);
                iconClass = 'fa-file-code text-green-400';
                fileType = 'text';
            }

            if (!content || content.trim().length === 0) {
                this.showToast('Не удалось извлечь текст из файла или файл пуст', 'error');
                return;
            }

            // Ограничиваем размер текста (чтобы не превысить лимиты LLM)
            const maxChars = 15000;
            if (content.length > maxChars) {
                this.log(`⚠️ Текст слишком длинный (${content.length} символов), обрезаем до ${maxChars}`, 'warning');
                content = content.substring(0, maxChars) + '\n\n[... текст обрезан ...]';
            }

            this.state.currentFile = file;
            this.state.fileContent = content;

            document.getElementById('fileInfo').classList.remove('hidden');
            document.getElementById('fileName').textContent = file.name;
            document.getElementById('fileSize').textContent = `${this.formatFileSize(file.size)} • ${content.length} символов`;
            
            const fileIcon = document.getElementById('fileIcon');
            fileIcon.className = `fas ${iconClass}`;
            
            // Показываем начало текста для проверки
            const preview = content.substring(0, 60).replace(/\n/g, ' ') + '...';
            document.getElementById('filePreview').textContent = preview;
            
            this.log(`✅ Файл загружен: ${fileType.toUpperCase()}, ${content.length} символов`, 'success');
            this.log(`📝 Превью: ${preview}`, 'info');
            
        } catch (error) {
            this.log(`❌ Ошибка обработки файла: ${error.message}`, 'error');
            this.showToast('Ошибка чтения файла: ' + error.message, 'error');
        }
    }

    // Извлечение текста из PDF
    async extractTextFromPDF(file) {
        const arrayBuffer = await file.arrayBuffer();
        const pdf = await pdfjsLib.getDocument({ data: arrayBuffer }).promise;
        
        let fullText = '';
        const totalPages = pdf.numPages;
        
        this.log(`📖 PDF: ${totalPages} страниц`, 'info');

        for (let i = 1; i <= totalPages; i++) {
            const page = await pdf.getPage(i);
            const textContent = await page.getTextContent();
            const pageText = textContent.items.map(item => item.str).join(' ');
            fullText += pageText + '\n';
            
            if (i % 5 === 0) {
                this.log(`⏳ Обработано страниц: ${i}/${totalPages}`, 'info');
            }
        }

        // Очищаем текст от лишних пробелов
        fullText = fullText.replace(/\s+/g, ' ').replace(/\n\s*\n/g, '\n\n').trim();
        
        return fullText;
    }

    // Извлечение текста из DOCX
    async extractTextFromDOCX(file) {
        const arrayBuffer = await file.arrayBuffer();
        const result = await mammoth.extractRawText({ arrayBuffer });
        
        if (result.messages.length > 0) {
            console.log('Mammoth messages:', result.messages);
        }

        // Очищаем текст
        let text = result.value;
        text = text.replace(/\s+/g, ' ').replace(/\n\s*\n/g, '\n\n').trim();
        
        return text;
    }

    formatFileSize(bytes) {
        if (bytes === 0) return '0 Bytes';
        const k = 1024;
        const sizes = ['Bytes', 'KB', 'MB', 'GB'];
        const i = Math.floor(Math.log(bytes) / Math.log(k));
        return parseFloat((bytes / Math.pow(k, i)).toFixed(2)) + ' ' + sizes[i];
    }

    saveRubric() {
        const name = document.getElementById('rubricName').value.trim();
        if (!name) {
            this.showToast('Введите название рубрики', 'error');
            return;
        }

        const items = [];
        document.querySelectorAll('#rubricItems > div').forEach(div => {
            const name = div.querySelector('[data-name]').value.trim();
            const desc = div.querySelector('[data-desc]').value.trim();
            const weight = parseInt(div.querySelector('[data-weight]').value) || 0;
            const maxScore = parseInt(div.querySelector('[data-max]').value) || 10;
            if (name) items.push({ name, description: desc, weight, maxScore });
        });

        if (items.length === 0) {
            this.showToast('Добавьте хотя бы один критерий', 'error');
            return;
        }

        const totalWeight = items.reduce((sum, item) => sum + item.weight, 0);
        if (totalWeight !== 100) {
            this.showToast(`Сумма весов должна быть 100%, сейчас ${totalWeight}%`, 'error');
            return;
        }

        const rubric = {
            id: 'rubric_' + Date.now(),
            name,
            criteria: items
        };

        this.state.rubrics.push(rubric);
        this.saveRubrics();
        this.loadRubrics();
        this.closeRubricBuilder();
        this.showToast('Рубрика сохранена', 'success');
        document.getElementById('rubricName').value = '';
    }

    async startAnalysis() {
        if (!this.state.fileContent) {
            this.showToast('Загрузите файл для проверки', 'error');
            return;
        }
        if (!this.state.selectedModel) {
            this.showToast('Выберите модель ИИ', 'error');
            return;
        }
        if (!this.state.selectedRubric) {
            this.showToast('Выберите рубрику оценки', 'error');
            return;
        }

        this.state.isAnalyzing = true;
        this.state.expertResults = {};
        this.state.evaluationPlan = null;
        
        document.getElementById('startBtn').disabled = true;
        document.getElementById('progressPanel').classList.remove('hidden');

        this.updateProgress(0);
        this.renderExpertSteps();

        try {
            this.log('📖 Начинаем двухэтапную проверку...', 'info');
            this.log(`📄 Размер текста: ${this.state.fileContent.length} символов`, 'info');

            // ЭТАП 1: Планировщик анализирует рубрику и формирует промты
            this.setExpertStatus('planner', 'running');
            this.log('🧠 Этап 1: Планировщик формирует задания для экспертов...', 'info');
            
            const plan = await this.runPlanner();
            this.state.evaluationPlan = plan;
            
            this.setExpertStatus('planner', 'completed', {score: 1, maxScore: 1});
            this.log('✅ Планировщик создал задания для критериев', 'success');
            this.updateProgress(15);

            // ЭТАП 2: Последовательная оценка каждого критерия
            const criteria = this.state.selectedRubric.criteria;
            const totalCriteria = criteria.length;
            
            for (let i = 0; i < totalCriteria; i++) {
                const criterion = criteria[i];
                const progress = 15 + ((i + 1) / totalCriteria) * 70; // 15% до 85%
                
                this.log(`🔍 Оцениваем: ${criterion.name}...`, 'info');
                
                // Находим соответствующий промт из плана
                const planItem = plan.find(p => 
                    p.criterion.toLowerCase().includes(criterion.name.toLowerCase()) ||
                    criterion.name.toLowerCase().includes(p.criterion.toLowerCase())
                );
                
                if (!planItem) {
                    this.log(`⚠️ Нет промта для ${criterion.name}, используем стандартный`, 'warning');
                }
                
                try {
                    const result = await this.evaluateCriterion(criterion, planItem);
                    this.state.expertResults[criterion.name] = result;
                    this.log(`✅ ${criterion.name}: ${result.score}/${criterion.maxScore} баллов`, 'success');
                } catch (error) {
                    this.log(`❌ Ошибка оценки ${criterion.name}: ${error.message}`, 'error');
                    this.state.expertResults[criterion.name] = {
                        score: Math.floor(criterion.maxScore * 0.5),
                        maxScore: criterion.maxScore,
                        justification: 'Ошибка оценки: ' + error.message,
                        recommendations: 'Проверьте вручную',
                        strengths: [],
                        weaknesses: []
                    };
                }
                
                this.updateProgress(progress);
            }

            // Финализация
            this.updateProgress(90);
            this.log('📝 Формирование итогового отчета...', 'info');

            const finalResult = this.calculateFinalScore();
            this.state.currentResult = finalResult;

            this.updateProgress(100);
            this.log(`🎉 Анализ завершен! Итоговая оценка: ${finalResult.finalScore} из 100`, 'success');

            setTimeout(() => this.showResults(), 500);

        } catch (error) {
            this.log(`❌ Критическая ошибка: ${error.message}`, 'error');
            this.showToast('Ошибка при проверке: ' + error.message, 'error');
        } finally {
            this.state.isAnalyzing = false;
            document.getElementById('startBtn').disabled = false;
        }
    }

    // ЭТАП 1: Планировщик
    async runPlanner() {
        const rubric = this.state.selectedRubric;
        const content = this.state.fileContent;
        
        // Берем первые 4000 символов для анализа планировщиком
        const contentPreview = content.substring(0, 4000);
        
        const prompt = `Ты - Планировщик академической проверки. Твоя задача - проанализировать текст работы и рубрику, и создать детальные промты для оценки каждого критерия.

РУБРИКА:
${rubric.criteria.map(c => `- ${c.name} (макс ${c.maxScore} балл., вес ${c.weight}%): ${c.description}`).join('\n')}

ТЕКСТ РАБОТЫ (начало):
${contentPreview}

ЗАДАЧА:
Создай JSON массив, где для каждого критерия рубрики будет детальный промт для эксперта. Промт должен включать конкретные вопросы и указания, на что обращать внимание при оценке.

ФОРМАТ ОТВЕТА (строго JSON):
[
  {
    "criterion": "Название критерия",
    "prompt": "Детальный промт для эксперта. Включи: 1) Что проверять, 2) На что обращать внимание, 3) Какие ошибки искать",
    "maxScore": ${rubric.criteria[0]?.maxScore || 10}
  },
  ...
]`;

        const response = await this.callLLM(prompt, 3000);
        
        // Парсим план
        try {
            const plan = this.extractJSON(response);
            if (Array.isArray(plan) && plan.length > 0) {
                this.log(`📋 Планировщик создал ${plan.length} заданий`, 'info');
                return plan;
            } else {
                throw new Error('Планировщик вернул некорректную структуру');
            }
        } catch (e) {
            this.log('⚠️ Не удалось распарсить план, используем стандартные промты', 'warning');
            // Возвращаем стандартные промты на основе рубрики
            return rubric.criteria.map(c => ({
                criterion: c.name,
                prompt: `Оцени критерий "${c.name}" (${c.description}). Проверь текст работы и выстави оценку от 0 до ${c.maxScore}.`,
                maxScore: c.maxScore
            }));
        }
    }

    // ЭТАП 2: Оценка конкретного критерия
    async evaluateCriterion(criterion, planItem) {
        const content = this.state.fileContent;
        const maxLen = 6000; // Отправляем больше текста для конкретной оценки
        const contentPreview = content.substring(0, maxLen);
        
        // Используем промт от планировщика или стандартный
        const expertPrompt = planItem ? planItem.prompt : `Оцени критерий "${criterion.name}"`;
        
        const prompt = `Ты - Эксперт по оценке академических работ. Оцени работу строго по указанному критерию.

КРИТЕРИЙ: ${criterion.name}
ОПИСАНИЕ: ${criterion.description || ''}
МАКСИМАЛЬНЫЙ БАЛЛ: ${criterion.maxScore}

ЗАДАНИЕ ДЛЯ ЭКСПЕРТА:
${expertPrompt}

ТЕКСТ РАБОТЫ ДЛЯ АНАЛИЗА:
---
${contentPreview}
---

ВЕРНИ РЕЗУЛЬТАТ СТРОГО В JSON:
{
  "score": число от 0 до ${criterion.maxScore},
  "justification": "развернутое обоснование оценки на русском языке",
  "recommendations": "конкретные рекомендации по улучшению",
  "strengths": ["сильная сторона 1", "сильная сторона 2"],
  "weaknesses": ["слабая сторона 1", "слабая сторона 2"]
}`;

        const response = await this.callLLM(prompt, 2000);
        
        // Парсим результат
        const result = this.parseExpertResponse(response, criterion);
        return result;
    }

    // Универсальный вызов LLM
    async callLLM(prompt, maxTokens = 2000) {
        const controller = new AbortController();
        const timeoutId = setTimeout(() => controller.abort(), 120000);

        try {
            const response = await fetch(`${this.config.lmStudioUrl}/v1/chat/completions`, {
                method: 'POST',
                headers: { 
                    'Content-Type': 'application/json',
                    'Accept': 'application/json'
                },
                body: JSON.stringify({
                    model: this.state.selectedModel,
                    messages: [
                        {
                            role: 'system',
                            content: 'Ты - академический эксперт. Отвечай только в формате JSON без markdown, без текста до или после JSON.'
                        },
                        { role: 'user', content: prompt }
                    ],
                    temperature: 0.3,
                    max_tokens: maxTokens
                }),
                signal: controller.signal
            });

            clearTimeout(timeoutId);

            if (!response.ok) {
                const errorText = await response.text();
                throw new Error(`HTTP ${response.status}: ${errorText}`);
            }

            const data = await response.json();
            return data.choices[0].message.content;
            
        } catch (error) {
            clearTimeout(timeoutId);
            throw error;
        }
    }

    // Извлечение JSON из ответа
    extractJSON(text) {
        // Пробуем найти JSON массив или объект
        const patterns = [
            /\[[\s\S]*\]/, // Массив
            /\{[\s\S]*\}/  // Объект
        ];
        
        for (const pattern of patterns) {
            const match = text.match(pattern);
            if (match) {
                try {
                    return JSON.parse(match[0]);
                } catch (e) {
                    continue;
                }
            }
        }
        
        throw new Error('JSON не найден');
    }

    parseExpertResponse(rawResponse, criterion) {
        // Пробуем извлечь JSON
        try {
            const data = this.extractJSON(rawResponse);
            
            let score = parseFloat(data.score);
            if (isNaN(score)) score = 0;
            if (score < 0) score = 0;
            if (score > criterion.maxScore) score = criterion.maxScore;
            
            return {
                score: score,
                maxScore: criterion.maxScore,
                justification: data.justification || data.comment || 'Нет обоснования',
                recommendations: data.recommendations || data.recommendation || 'Нет рекомендаций',
                strengths: Array.isArray(data.strengths) ? data.strengths : [],
                weaknesses: Array.isArray(data.weaknesses) ? data.weaknesses : (Array.isArray(data.weakness) ? data.weakness : [])
            };
        } catch (e) {
            // Fallback: ищем числа в тексте
            const scoreMatch = rawResponse.match(/(\d+(?:\.\d+)?)\s*\/\s*(\d+)/);
            if (scoreMatch) {
                const score = parseFloat(scoreMatch[1]);
                return {
                    score: Math.min(score, criterion.maxScore),
                    maxScore: criterion.maxScore,
                    justification: 'Оценка извлечена из текста: ' + rawResponse.substring(0, 200),
                    recommendations: 'См. полный ответ модели',
                    strengths: [],
                    weaknesses: []
                };
            }
            
            return {
                score: Math.floor(criterion.maxScore * 0.5),
                maxScore: criterion.maxScore,
                justification: 'Не удалось распарсить ответ модели',
                recommendations: 'Проверьте вручную',
                strengths: [],
                weaknesses: []
            };
        }
    }

    calculateFinalScore() {
        const rubric = this.state.selectedRubric;
        const results = this.state.expertResults;

        let totalWeightedScore = 0;
        let totalWeight = 0;
        const criterionResults = [];

        rubric.criteria.forEach(criterion => {
            const expertResult = results[criterion.name] || {
                score: 0,
                maxScore: criterion.maxScore,
                justification: 'Не оценено',
                recommendations: '-',
                strengths: [],
                weaknesses: []
            };

            const normalizedScore = (expertResult.score / criterion.maxScore) * 100;
            const weightedScore = normalizedScore * (criterion.weight / 100);

            totalWeightedScore += weightedScore;
            totalWeight += criterion.weight;

            criterionResults.push({
                criterion: criterion.name,
                weight: criterion.weight,
                score: expertResult.score,
                maxScore: criterion.maxScore,
                normalizedScore: Math.round(normalizedScore * 10) / 10,
                weightedScore: Math.round(weightedScore * 10) / 10,
                justification: expertResult.justification,
                recommendations: expertResult.recommendations,
                strengths: expertResult.strengths,
                weaknesses: expertResult.weaknesses
            });
        });

        const finalScore = totalWeight > 0 ? Math.round((totalWeightedScore / totalWeight) * 100) / 1 : 0;

        let grade, gradeDescription;
        if (finalScore >= 90) { grade = 'A'; gradeDescription = 'Отлично'; }
        else if (finalScore >= 80) { grade = 'B'; gradeDescription = 'Хорошо'; }
        else if (finalScore >= 70) { grade = 'C'; gradeDescription = 'Удовлетворительно'; }
        else if (finalScore >= 60) { grade = 'D'; gradeDescription = 'Достаточно'; }
        else if (finalScore >= 50) { grade = 'E'; gradeDescription = 'Слабо'; }
        else { grade = 'F'; gradeDescription = 'Неудовлетворительно'; }

        return {
            finalScore: Math.round(finalScore),
            maxScore: 100,
            percentage: Math.round(finalScore),
            grade: grade,
            gradeDescription: gradeDescription,
            fileName: this.state.currentFile.name,
            timestamp: new Date().toISOString(),
            criteria: criterionResults,
            rubricName: rubric.name,
            modelUsed: this.state.selectedModel
        };
    }

        const element = document.getElementById(id === 'planner' ? 'expert-planner' : `expert-${id}`);

        element.querySelector('.expert-loading')?.classList.add('hidden');
        element.querySelector('.expert-done')?.classList.add('hidden');

        if (status === 'running') {
            element.classList.add('border-indigo-500/50', 'bg-indigo-500/10');
            element.querySelector('.expert-status').textContent = 'Анализирует...';
            element.querySelector('.expert-loading')?.classList.remove('hidden');

            element.classList.add('border-green-500/50', 'bg-green-500/10');
            element.querySelector('.expert-status').textContent = 'Завершено';
            element.querySelector('.expert-status').classList.add('text-green-400');
            element.querySelector('.expert-done')?.classList.remove('hidden');

            if (result && result.score !== undefined) {
                const scoreEl = element.querySelector('.expert-score');
                if (scoreEl) {
                    scoreEl.textContent = `${result.score}/${result.maxScore} баллов`;
                    scoreEl.classList.remove('hidden');
                }
            }
        }
    }

        localStorage.setItem('history', JSON.stringify(this.state.history));
        this.loadHistory();
        this.showToast('Сохранено в историю', 'success');
    }

    loadHistory() {
        const container = document.getElementById('historyList');
        if (this.state.history.length === 0) {
            container.innerHTML = '<div class="text-center text-slate-500 py-8 text-sm">История пуста</div>';
            return;
        }

        container.innerHTML = this.state.history.map((item, idx) => `
            <div class="p-3 rounded-lg bg-slate-800/30 border border-slate-700 hover:border-indigo-500/50 cursor-pointer transition-colors"
                 onclick="app.loadHistoryItem(${idx})">
                <div class="flex items-center justify-between mb-1">
                    <span class="font-medium text-sm truncate">${item.fileName}</span>
                    <span class="text-indigo-400 font-bold">${item.finalScore}</span>
                </div>
                <div class="flex items-center justify-between text-xs text-slate-500">
                    <span>${new Date(item.timestamp).toLocaleDateString()}</span>
                    <span>${item.grade}</span>
                </div>
            </div>
        `).join('');

        document.getElementById('statTotal').textContent = this.state.history.length;

        document.getElementById('statAvg').textContent = avg ? avg.toFixed(1) : '-';
    }
