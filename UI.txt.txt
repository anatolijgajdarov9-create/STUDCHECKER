            setupEventListeners() {
                // Drag & Drop
                ['dragenter', 'dragover', 'dragleave', 'drop'].forEach(eventName => {
                    this.ui.dropZone.addEventListener(eventName, (e) => {
                        e.preventDefault();
                        e.stopPropagation();
                    });
                });

                this.ui.dropZone.addEventListener('dragover', () => this.ui.dropZone.classList.add('dragover'));
                this.ui.dropZone.addEventListener('dragleave', () => this.ui.dropZone.classList.remove('dragover'));
                this.ui.dropZone.addEventListener('drop', (e) => {
                    this.ui.dropZone.classList.remove('dragover');
                    const files = e.dataTransfer.files;
                    if (files.length) this.handleFile(files[0]);
                });

                this.ui.fileInput.addEventListener('change', (e) => {
                    if (e.target.files.length) this.handleFile(e.target.files[0]);

            async checkConnection() {
                try {
                    const response = await fetch(`${this.config.url}/models`, { method: 'GET' });
                    if (response.ok) {
                        const data = await response.json();
                        this.state.connected = true;
                        this.state.models = data.data || [];

            async handleFile(file) {
                this.ui.filePreview.classList.remove('hidden');
                this.ui.fileName.textContent = file.name;
                this.ui.fileSize.textContent = (file.size / 1024).toFixed(1) + ' KB';
                
                // Icon logic
                const ext = file.name.split('.').pop().toLowerCase();
                const icons = { pdf: 'fa-file-pdf text-red-500', docx: 'fa-file-word text-blue-600', txt: 'fa-file-lines text-slate-500', md: 'fa-file-code text-slate-700' };
                const defaultIcon = 'fa-file';
                const iconClass = icons[ext] || defaultIcon;
                this.ui.fileIcon.className = `fa-solid ${iconClass.split(' ')[0]} ${iconClass.split(' ')[1] || ''}`;

                this.state.currentFile = file;
                this.log(`Файл загружен: ${file.name}`, 'info');
                
                // Extract text (Mock for demo if libraries fail or for speed)
                try {
                    if (ext === 'pdf') {
                        const arrayBuffer = await file.arrayBuffer();
                        const pdf = await pdfjsLib.getDocument(arrayBuffer).promise;
                        let text = '';
                        for (let i = 1; i <= pdf.numPages; i++) {
                            const page = await pdf.getPage(i);
                            text += (await page.getTextContent()).items.map(item => item.str).join(' ');
                        }
                        this.state.fileContent = text.substring(0, 15000);
                    } else if (ext === 'docx') {
                        const arrayBuffer = await file.arrayBuffer();
                        const result = await mammoth.extractRawText({ arrayBuffer: arrayBuffer });
                        this.state.fileContent = result.value.substring(0, 15000);
                    } else {
                        const text = await file.text();
                        this.state.fileContent = text.substring(0, 15000);
                    }
                    
                    if (this.state.fileContent.length >= 15000) {
                        this.log('Текст обрезан до 15000 символов', 'warning');
                    }
                    this.log(`Извлечено ${this.state.fileContent.length} символов`, 'success');
                } catch (e) {
                    this.log('Ошибка чтения файла. Используем заглушку.', 'error');
                    this.state.fileContent = "Это пример текста студенческой работы для демонстрации работы системы Smart Grader AI. Работа посвящена анализу современных методов машинного обучения...";
                }
            }

            async runExperts() {

                // Reset progress UI
                this.ui.expertSteps.innerHTML = '';

                    div.className = 'flex items-center gap-3 p-3 bg-slate-50 rounded-lg border border-slate-100 opacity-50 transition-all';

                            </div>
                        </div>
                    `;
                    this.ui.expertSteps.appendChild(div);
                });

                // Process each criterion
                for (let i = 0; i < rubric.criteria.length; i++) {
                    const criterion = rubric.criteria[i];
                    const stepEl = document.getElementById(`step-${i}`);
                    const statusText = stepEl.querySelector('.status-text');
                    const bar = stepEl.querySelector('.bg-blue-500');

                    stepEl.classList.remove('opacity-50');

                    statusText.classList.add('text-blue-600');
                    bar.style.width = '50%';

                    // Call LLM (Mock)
                    const result = await this.mockLLMCall(criterion);

                    bar.style.width = '100%';

                    statusText.classList.remove('text-blue-600');

                    this.ui.progressPercent.textContent = `${percent}%`;

                    this.log(`Критерий "${criterion.name}" оценен: ${result.score}/${criterion.max}`, 'info');
                    await new Promise(r => setTimeout(r, 800)); // Delay between steps
                }

                    grade: grade,
                    details: results

                document.getElementById('resGrade').textContent = grade;

                this.updateStats();
                this.closeModal('resultsModal');
                this.showToast('Результат сохранен в историю', 'success');
            }

            renderHistory() {
                const list = this.ui.historyList;
                if (!this.state.history.length) {
                    list.innerHTML = '<div class="text-center py-8 text-slate-400 text-sm"><i class="fa-regular fa-folder-open text-2xl mb-2 opacity-50"></i><p>История пуста</p></div>';
                    return;
                }

                list.innerHTML = this.state.history.map((h, i) => `
                    <div class="p-3 bg-slate-50 hover:bg-blue-50 rounded-lg border border-slate-100 cursor-pointer transition-colors group" onclick="app.loadHistoryItem(${i})">
                        <div class="flex justify-between items-start">
                            <div>
                                <p class="text-sm font-medium text-slate-800 truncate w-40">${h.fileName}</p>
                                <p class="text-xs text-slate-500">${new Date(h.date).toLocaleDateString()}</p>
                            </div>
                            <div class="text-right">
                                <span class="block text-lg font-bold ${h.grade === 'A' || h.grade === 'B' ? 'text-green-600' : 'text-slate-700'}">${h.score}%</span>
                                <span class="text-xs font-bold bg-white border border-slate-200 px-1.5 rounded text-slate-600">${h.grade}</span>
                            </div>
                        </div>
                    </div>
                `).join('');
            }

            loadHistoryItem(index) {
                const item = this.state.history[index];
                this.state.currentResult = item;
                // Re-render modal with saved data (simplified for demo)
                this.showResultsModal(); 
                // Note: In a real app, we'd reconstruct the chart and list from saved data specifically
            }

            updateStats() {
                this.ui.statCount.textContent = this.state.history.length;
                if (this.state.history.length > 0) {
                    const avg = Math.round(this.state.history.reduce((acc, curr) => acc + curr.score, 0) / this.state.history.length);
                    this.ui.statAvg.textContent = avg;
                } else {
                    this.ui.statAvg.textContent = '-';
                }
            }

            downloadReport() {
                const r = this.state.currentResult;
                const content = `# Отчет Smart Grader AI\n\nФайл: ${r.fileName}\nДата: ${new Date(r.date).toLocaleString()}\nОценка: ${r.score}% (${r.grade})\n\n## Детали\n${r.details.map(d => `- ${d.criterionName}: ${d.score}/${d.maxScore}`).join('\n')}`;
                const blob = new Blob([content], { type: 'text/markdown' });
                const url = URL.createObjectURL(blob);
                const a = document.createElement('a');
                a.href = url;
                a.download = `report_${r.fileName.split('.')[0]}.md`;
                a.click();
            }

            // --- Modals & Utils ---

            openModal(id) {
                document.getElementById(id).classList.remove('hidden');
                if (id === 'rubricModal') this.initRubricBuilder();
            }

            closeModal(id) {
                document.getElementById(id).classList.add('hidden');
            }

            log(msg, type = 'info') {
                const time = new Date().toLocaleTimeString();
                // Light theme colors
                const colors = { 
                    info: 'text-blue-600', 
                    success: 'text-green-600', 
                    error: 'text-red-600', 
                    warning: 'text-orange-600' 
                };
                const icons = { 
                    info: 'fa-info-circle', 
                    success: 'fa-check-circle', 
                    error: 'fa-exclamation-circle', 
                    warning: 'fa-triangle-exclamation' 
                };
                
                const div = document.createElement('div');
                div.className = `log-entry ${colors[type]}`;
                div.innerHTML = `<span class="log-time">[${time}]</span><i class="fa-solid ${icons[type]} log-icon"></i> <span class="font-medium">${msg}</span>`;
                this.ui.logContent.appendChild(div);
                this.ui.logContent.scrollTop = this.ui.logContent.scrollHeight;
            }

            showToast(msg, type = 'info') {
                const container = document.getElementById('toastContainer');
                const colors = { success: 'bg-green-500', error: 'bg-red-500', info: 'bg-blue-500', warning: 'bg-orange-500' };
                const icons = { success: 'fa-check', error: 'fa-xmark', info: 'fa-info', warning: 'fa-exclamation' };
                
                const toast = document.createElement('div');
                toast.className = `toast toast-hidden flex items-center gap-3 px-4 py-3 rounded-lg shadow-lg text-white ${colors[type]}`;
                toast.innerHTML = `<i class="fa-solid ${icons[type]}"></i> <span class="font-medium text-sm">${msg}</span>`;
                
                container.appendChild(toast);
                
                // Animate in
                requestAnimationFrame(() => toast.classList.remove('toast-hidden'));
                
                // Remove
                setTimeout(() => {
                    toast.classList.add('toast-hidden');
                    setTimeout(() => toast.remove(), 300);
                }, 4000);
            }

            // --- Rubric Builder Logic ---
            
            initRubricBuilder() {
                document.getElementById('criteriaBuilder').innerHTML = '';
                document.getElementById('newRubricName').value = '';
                this.addCriterionRow(); // Add one empty row
                this.updateTotalWeight();
            }

            addCriterionRow(data = { name: '', weight: 20, max: 10 }) {
                const container = document.getElementById('criteriaBuilder');
                const div = document.createElement('div');
                div.className = 'flex gap-2 items-end bg-slate-50 p-2 rounded border border-slate-200';
                div.innerHTML = `
                    <div class="flex-1">
                        <label class="text-xs text-slate-500">Название</label>
                        <input type="text" value="${data.name}" class="crit-name w-full text-sm border-b border-slate-300 bg-transparent focus:border-blue-500 outline-none py-1" placeholder="Критерий...">
                    </div>
                    <div class="w-20">
                        <label class="text-xs text-slate-500">Вес (%)</label>
                        <input type="number" value="${data.weight}" class="crit-weight w-full text-sm border-b border-slate-300 bg-transparent focus:border-blue-500 outline-none py-1 text-center" onchange="app.updateTotalWeight()">
                    </div>
                    <div class="w-20">
                        <label class="text-xs text-slate-500">Балл</label>
                        <input type="number" value="${data.max}" class="crit-max w-full text-sm border-b border-slate-300 bg-transparent focus:border-blue-500 outline-none py-1 text-center">
                    </div>
                    <button onclick="this.parentElement.remove(); app.updateTotalWeight()" class="mb-1 text-slate-400 hover:text-red-500"><i class="fa-solid fa-trash"></i></button>
                `;
                container.appendChild(div);
            }

            updateTotalWeight() {
                const weights = document.querySelectorAll('.crit-weight');
                let total = 0;
                weights.forEach(w => total += parseInt(w.value) || 0);
                const el = document.getElementById('totalWeight');
                el.textContent = total + '%';
                el.className = total === 100 ? 'font-bold text-green-600' : 'font-bold text-red-500';
            }

            saveNewRubric() {
                const name = document.getElementById('newRubricName').value;
                if (!name) return this.showToast('Введите название рубрики', 'error');
                
                const weights = document.querySelectorAll('.crit-weight');
                let total = 0;
                weights.forEach(w => total += parseInt(w.value) || 0);
                if (total !== 100) return this.showToast('Сумма весов должна быть 100%', 'error');

                const criteria = [];
                const rows = document.querySelectorAll('#criteriaBuilder > div');
                rows.forEach(row => {
                    criteria.push({
                        id: 'c-' + Date.now() + Math.random(),
                        name: row.querySelector('.crit-name').value,
                        weight: parseInt(row.querySelector('.crit-weight').value),
                        max: parseInt(row.querySelector('.crit-max').value)
                    });
                });

                const newRubric = { id: 'r-' + Date.now(), name, criteria };
                this.state.rubrics.push(newRubric);
                localStorage.setItem('sg_rubrics', JSON.stringify(this.state.rubrics));
                
                this.renderRubricSelect();
                this.selectRubric(newRubric.id);
                this.closeModal('rubricModal');
                this.showToast('Рубрика сохранена', 'success');
            }

            saveSettings() {
                this.config.url = document.getElementById('settingUrl').value;
                this.config.temp = parseFloat(document.getElementById('settingTemp').value);
                this.config.maxTokens = parseInt(document.getElementById('settingTokens').value);
                
                localStorage.setItem('sg_url', this.config.url);
                localStorage.setItem('sg_temp', this.config.temp);
                localStorage.setItem('sg_tokens', this.config.maxTokens);
                
                this.closeModal('settingsModal');
                this.showToast('Настройки сохранены', 'success');
                this.checkConnection(); // Re-check connection with new URL
            }
        }
