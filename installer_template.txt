                }
            });

            footerInfo.textContent = `Шаг ${currentStep} из ${totalSteps}`;
            btnBack.disabled = currentStep === 1 || isInstalling;
            
            if (currentStep === 4) {
                btnNext.textContent = "Закрыть";
                btnNext.classList.add('btn-success');
                btnNext.classList.remove('btn-primary');
                btnBack.style.display = 'none';
            } else if (currentStep === 2) {
                btnNext.disabled = !licenseAccepted;
                btnNext.textContent = "Установить";
            } else {
                btnNext.textContent = "Далее";
                btnNext.disabled = false;
            }
        }

        // Navigation
        function goToStep(step) {
            if (isInstalling && step === 2) return;
            currentStep = step;
            updateStepIndicator();

            if (currentStep === 3) {
                startInstallation();
            }
        }

        btnNext.addEventListener('click', () => {
            if (currentStep === 4) {
                if (launchOnFinish) {
                    alert("Запуск системы автоматизированной проверки студенческих работ...");
                }
                document.querySelector('.installer-window').style.animation = 'fadeOut 0.3s ease forwards';
                setTimeout(() => {
                    document.body.innerHTML = '<div style="text-align:center;color:#666;">Приложение закрыто</div>';
                }, 300);
                return;
            }
            goToStep(currentStep + 1);
        });

        btnBack.addEventListener('click', () => {
            if (currentStep > 1 && !isInstalling) {
                goToStep(currentStep - 1);
            }
        });

        // License checkbox
        licenseCheckbox.addEventListener('click', () => {
            licenseAccepted = !licenseAccepted;
            licenseCheckbox.querySelector('.checkbox').classList.toggle('checked', licenseAccepted);
            btnNext.disabled = !licenseAccepted;
        });

        // Launch checkbox
        launchCheckbox.addEventListener('click', () => {
            launchOnFinish = !launchOnFinish;
            launchCheckbox.querySelector('.checkbox').classList.toggle('checked', launchOnFinish);
        });

        // Browse button
        btnBrowse.addEventListener('click', () => {
            // Simulate folder selection
            const paths = [
                "C:\\Program Files\\САПСР",
                "C:\\Apps\\САПСР",
                "D:\\Software\\САПСР",
                "C:\\Users\\Admin\\САПСР"
            ];
            const randomPath = paths[Math.floor(Math.random() * paths.length)];
            installPath.value = randomPath;
        });

        // Installation simulation
        function formatTime(seconds) {
            const m = Math.floor(seconds / 60);
            const s = seconds % 60;
            return `${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`;
        }

        function addFileEntry(fileName) {
            const div = document.createElement('div');
            div.className = 'file-item';
            div.innerHTML = `
                <svg viewBox="0 0 24 24"><path d="M14 2H6c-1.1 0-1.99.9-1.99 2L4 20c0 1.1.89 2 1.99 2H18c1.1 0 2-.9 2-2V8l-6-6zm2 16H8v-2h8v2zm0-4H8v-2h8v2zm-3-5V3.5L18.5 9H13z"/></svg>
                <span>${fileName}</span>
            `;
            fileList.insertBefore(div, fileList.firstChild);
            
            // Keep only last 5 items
            while (fileList.children.length > 5) {
                fileList.removeChild(fileList.lastChild);
            }
        }

        function startInstallation() {
            isInstalling = true;
            btnBack.disabled = true;
            btnNext.disabled = true;
            
            let progress = 0;
            const startTime = Date.now();
            const totalDuration = 8000; // 8 seconds

            function animate() {
                if (!isInstalling) return;

                const elapsed = Date.now() - startTime;
                progress = Math.min(100, (elapsed / totalDuration) * 100);

                // Find current step
                const currentInstallStep = installSteps.slice().reverse().find(step => progress >= step.threshold);
                
                if (currentInstallStep) {
                    statusText.textContent = currentInstallStep.text;
                    addFileEntry(currentInstallStep.file);
                }

                progressBar.style.width = `${progress}%`;
                percentage.textContent = `${Math.floor(progress)}%`;

                const remainingSeconds = Math.max(0, Math.ceil((totalDuration - elapsed) / 1000));
                timeRemaining.textContent = `Осталось: ${formatTime(remainingSeconds)}`;

                if (progress < 100) {
                    requestAnimationFrame(animate);
                } else {
                    // Installation complete
                    setTimeout(() => {
                        finalPath.textContent = installPath.value;
                        const now = new Date();
                        installDate.textContent = now.toLocaleDateString('ru-RU');
                        goToStep(4);
                    }, 500);
                }
            }

            animate();
        }

        // Initialize
        updateStepIndicator();

        // Add fade out animation
        const style = document.createElement('style');
        style.textContent = `
            @keyframes fadeOut {
                from { opacity: 1; transform: scale(1); }
                to { opacity: 0; transform: scale(0.95); }
            }
        `;
        document.head.appendChild(style);
    </script>
